FUNC cip_product_line_visible:
CREATE OR REPLACE FUNCTION public.cip_product_line_visible(line text)
 RETURNS boolean
 LANGUAGE sql
 STABLE
AS $function$
          SELECT CASE
            WHEN coalesce(current_setting('cip.product_lines', true), '') IN ('', '*') THEN true
            WHEN current_setting('cip.product_lines', true) = '!' THEN false
            ELSE line IS NOT NULL
                 AND line = ANY (string_to_array(current_setting('cip.product_lines', true), ','))
          END
        $function$
